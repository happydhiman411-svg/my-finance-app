// ============================================================ // FINANCE MANAGER - APPS SCRIPT BACKEND (SECURE) // ============================================================ var SHEET_TRANS = "Transactions"; var SHEET_CAT = "Categories"; var SHEET_SETTINGS = "Settings"; var SHEET_DAILY = "DailyLedger"; var SHEET_AUTH = "_AuthTokens"; var TRANS_HEADERS = ["ID", "DATE", "TYPE", "CATEGORY", "PERSON", "AMOUNT", "PAYMENT MODE", "NOTE"]; var SALT = "fin_manager_salt_v2_2026"; // change if you want var TOKEN_DAYS = 30; var DEFAULT_USER = "Amine"; var DEFAULT_PASS = "Admin123"; // ------------------------------------------------------------ // ROUTER // ------------------------------------------------------------ function doGet(e) { return route(e.parameter || {}); } function doPost(e) { var data = {}; try { data = JSON.parse(e.postData.contents); } catch (err) {} return route(data); } function route(params) { var ss = SpreadsheetApp.getActiveSpreadsheet(); migrateSettings(ss); var action = params.action; // ---------- PUBLIC ---------- if (!action || action === "ping") return jsonRes({ status: "active" }); if (action === "verifyLogin") { return verifyLogin(ss, params.username, params.password); } // ---------- AUTH REQUIRED ---------- if (!isValidToken(ss, params.token)) { return jsonRes({ status: "error", code: "AUTH", message: "Session expired. Please login again." }); } switch (action) { case "getData": return getData(ss); case "addCategory": return addCategory(ss, params.name); case "saveTransaction": return saveTransaction(ss, params); case "saveBulk": return saveBulk(ss, params); case "deleteTransaction": return deleteTransaction(ss, params); case "verifyPassword": return jsonRes({ status: verifyPasswordOnly(ss, params.password) ? "success" : "error", message: verifyPasswordOnly(ss, params.password) ? "OK" : "Invalid password" }); case "updateCredentials": return updateCredentials(ss, params); case "logout": return logout(ss, params.token); } return jsonRes({ status: "error", message: "Invalid action" }); } // ------------------------------------------------------------ // AUTH // ------------------------------------------------------------ function hashPassword(p) { var digest = Utilities.computeDigest( Utilities.DigestAlgorithm.SHA_256, String(p) + SALT, Utilities.Charset.UTF_8 ); return digest.map(function (b) { return ('0' + (b & 0xFF).toString(16)).slice(-2); }).join(''); } function getStoredUser(ss) { var sheet = getOrCreateSheet(ss, SHEET_SETTINGS, ["Key", "Value"]); var rows = sheet.getDataRange().getValues(); for (var i = 0; i < rows.length; i++) { var k = String(rows[i][0]).toLowerCase().trim(); if (k === "user" || k === "username") return String(rows[i][1]).trim(); } sheet.appendRow(["User", DEFAULT_USER]); return DEFAULT_USER; } function getStoredPassHash(ss) { var sheet = getOrCreateSheet(ss, SHEET_SETTINGS, ["Key", "Value"]); var rows = sheet.getDataRange().getValues(); for (var i = 0; i < rows.length; i++) { var k = String(rows[i][0]).toLowerCase().trim(); if (k.indexOf("pass") !== -1) return String(rows[i][1]).trim(); } var h = hashPassword(DEFAULT_PASS); sheet.appendRow(["Password", h]); return h; } function migrateSettings(ss) { var sheet = getOrCreateSheet(ss, SHEET_SETTINGS, ["Key", "Value"]); var rows = sheet.getDataRange().getValues(); var hasUser = false, hasPass = false; for (var i = 0; i < rows.length; i++) { var k = String(rows[i][0]).toLowerCase().trim(); if (k === "user" || k === "username") hasUser = true; if (k.indexOf("pass") !== -1) { hasPass = true; var v = String(rows[i][1]).trim(); // Agar purani plain text password hai to hash me badlo if (!/^[a-f0-9]{64}$/.test(v)) { sheet.getRange(i + 1, 2).setValue(hashPassword(DEFAULT_PASS)); } } } if (!hasUser) sheet.appendRow(["User", DEFAULT_USER]); if (!hasPass) sheet.appendRow(["Password", hashPassword(DEFAULT_PASS)]); } function verifyLogin(ss, username, password) { var storedUser = getStoredUser(ss); var storedHash = getStoredPassHash(ss); if (String(username || "").trim() !== storedUser) { return jsonRes({ status: "error", message: "Invalid username or password" }); } if (hashPassword(password || "") !== storedHash) { return jsonRes({ status: "error", message: "Invalid username or password" }); } var token = Utilities.getUuid(); saveToken(ss, token); return jsonRes({ status: "success", token: token, user: storedUser }); } function verifyPasswordOnly(ss, password) { return hashPassword(password || "") === getStoredPassHash(ss); } function saveToken(ss, token) { var sheet = getOrCreateSheet(ss, SHEET_AUTH, ["Token", "Expiry"]); sheet.appendRow([token, Date.now() + TOKEN_DAYS * 24 * 60 * 60 * 1000]); // Cleanup expired var rows = sheet.getDataRange().getValues(); var keep = [["Token", "Expiry"]]; for (var i = 1; i < rows.length; i++) { if (Number(rows[i][1]) > Date.now()) keep.push(rows[i]); } if (keep.length !== rows.length) { sheet.clear(); sheet.getRange(1, 1, keep.length, 2).setValues(keep); } } function isValidToken(ss, token) { if (!token) return false; var sheet = ss.getSheetByName(SHEET_AUTH); if (!sheet) return false; var rows = sheet.getDataRange().getValues(); var now = Date.now(); for (var i = 1; i < rows.length; i++) { if (String(rows[i][0]) === String(token) && Number(rows[i][1]) > now) return true; } return false; } function logout(ss, token) { var sheet = ss.getSheetByName(SHEET_AUTH); if (!sheet) return jsonRes({ status: "success" }); var rows = sheet.getDataRange().getValues(); for (var i = 1; i < rows.length; i++) { if (String(rows[i][0]) === String(token)) { sheet.deleteRow(i + 1); break; } } return jsonRes({ status: "success" }); } function invalidateAllTokens(ss) { var sheet = ss.getSheetByName(SHEET_AUTH); if (sheet) { sheet.clear(); sheet.appendRow(["Token", "Expiry"]); } } // ------------------------------------------------------------ // DATA // ------------------------------------------------------------ function getData(ss) { return jsonRes({ status: "success", transactions: sheetToJSON(ss, SHEET_TRANS), categories: sheetToJSON(ss, SHEET_CAT), dailyLedger: sheetToJSON(ss, SHEET_DAILY) }); } function saveTransaction(ss, d) { var sheet = getOrCreateSheet(ss, SHEET_TRANS, TRANS_HEADERS); if (d.isEdit) { // Edit ke liye password zaroori if (!verifyPasswordOnly(ss, d.password)) { return jsonRes({ status: "error", message: "Invalid password" }); } var rows = sheet.getDataRange().getValues(); for (var i = 1; i < rows.length; i++) { if (String(rows[i][0]) === String(d.id)) { sheet.getRange(i + 1, 1, 1, 8).setValues([[ d.id, d.date, d.type, d.category, d.person, d.amount, d.paymentMode, d.note ]]); return jsonRes({ status: "success" }); } } return jsonRes({ status: "error", message: "Record not found" }); } sheet.appendRow([ d.id || ('t_' + Date.now()), d.date, d.type, d.category, d.person, d.amount, d.paymentMode, d.note ]); return jsonRes({ status: "success" }); } function saveBulk(ss, d) { var items = d.items || []; if (items.length === 0) return jsonRes({ status: "error", message: "No items" }); if (items.length > 500) return jsonRes({ status: "error", message: "Max 500 entries at once" }); var sheet = getOrCreateSheet(ss, SHEET_TRANS, TRANS_HEADERS); var rows = items.map(function (t, idx) { return [ 't_' + Date.now() + '_' + idx + '_' + Math.random().toString(36).slice(2, 6), t.date, t.type, t.category, t.person, t.amount, t.paymentMode, t.note ]; }); var startRow = sheet.getLastRow() + 1; sheet.getRange(startRow, 1, rows.length, 8).setValues(rows); return jsonRes({ status: "success", count: rows.length }); } function deleteTransaction(ss, d) { if (!verifyPasswordOnly(ss, d.password)) { return jsonRes({ status: "error", message: "Invalid password" }); } var sheet = ss.getSheetByName(SHEET_TRANS); if (!sheet) return jsonRes({ status: "error", message: "No data" }); var rows = sheet.getDataRange().getValues(); for (var i = 1; i < rows.length; i++) { if (String(rows[i][0]) === String(d.id)) { sheet.deleteRow(i + 1); return jsonRes({ status: "success" }); } } return jsonRes({ status: "error", message: "Record not found" }); } function addCategory(ss, name) { if (!name || !String(name).trim()) return jsonRes({ status: "error", message: "Empty name" }); var sheet = getOrCreateSheet(ss, SHEET_CAT, ["Category"]); var rows = sheet.getDataRange().getValues(); var val = String(name).trim(); for (var i = 1; i < rows.length; i++) { if (String(rows[i][0]).trim().toLowerCase() === val.toLowerCase()) { return jsonRes({ status: "error", message: "Category already exists" }); } } sheet.appendRow([val]); return jsonRes({ status: "success" }); } function updateCredentials(ss, d) { if (!verifyPasswordOnly(ss, d.currentPassword)) { return jsonRes({ status: "error", message: "Current password incorrect" }); } var sheet = getOrCreateSheet(ss, SHEET_SETTINGS, ["Key", "Value"]); var rows = sheet.getDataRange().getValues(); var changedPassword = false; // Change User if (d.newUsername && String(d.newUsername).trim()) { var uname = String(d.newUsername).trim(); var foundUser = false; for (var i = 0; i < rows.length; i++) { var k = String(rows[i][0]).toLowerCase().trim(); if (k === "user" || k === "username") { sheet.getRange(i + 1, 2).setValue(uname); foundUser = true; break; } } if (!foundUser) sheet.appendRow(["User", uname]); } // Change Password if (d.newPassword && String(d.newPassword).trim()) { var newHash = hashPassword(String(d.newPassword).trim()); var foundPass = false; for (var i = 0; i < rows.length; i++) { var k = String(rows[i][0]).toLowerCase().trim(); if (k.indexOf("pass") !== -1) { sheet.getRange(i + 1, 2).setValue(newHash); foundPass = true; break; } } if (!foundPass) sheet.appendRow(["Password", newHash]); changedPassword = true; } // Agar password badla to saare tokens invalid kar do (user ko dobara login karna hoga) if (changedPassword) invalidateAllTokens(ss); return jsonRes({ status: "success", relogin: changedPassword }); } // ------------------------------------------------------------ // HELPERS // ------------------------------------------------------------ function getOrCreateSheet(ss, name, headers) { var sheet = ss.getSheetByName(name); if (!sheet) { sheet = ss.insertSheet(name); if (headers) sheet.appendRow(headers); } return sheet; } function sheetToJSON(ss, name) { var sheet = ss.getSheetByName(name); if (!sheet) return []; var rows = sheet.getDataRange().getValues(); if (rows.length <= 1) return []; var rawHeaders = rows[0]; var tz = ss.getSpreadsheetTimeZone() || "Asia/Kolkata"; var data = []; for (var i = 1; i < rows.length; i++) { var obj = {}; for (var j = 0; j < rawHeaders.length; j++) { var h = String(rawHeaders[j]).trim().toLowerCase().replace(/\s+/g, ''); var val = rows[i][j]; if (val instanceof Date) { val = Utilities.formatDate(val, tz, "yyyy-MM-dd"); } else if (val !== null && val !== undefined) { val = String(val).trim(); } obj[h] = val; if (h === 'paymentmode') { obj['paymentMode'] = val; } if (h === 'category') { obj['category'] = val; } } data.push(obj); } return data; } function jsonRes(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
